### **Método create_password (Líneas 215-226)**
```python
@classmethod
def create_password(cls, password: str) -> str:
```
**QUÉ HACE:** Convierte contraseña en texto plano a hash seguro
**POR QUÉ @classmethod:** No necesita una instancia de usuario

```python
# Líneas 225-229
hashed_password = bcrypt.hashpw(
    password.encode('utf-8'), 
    bcrypt.gensalt(rounds=12)
)
return hashed_password.decode('utf-8')
```
**QUÉ HACE:** Crea hash bcrypt con salt aleatorio
**DESGLOSE:**
- `password.encode('utf-8')` → Convierte string a bytes
- `bcrypt.gensalt(rounds=12)` → Genera salt con complejidad 12
- `rounds=12` → Balance entre seguridad y velocidad
- `.decode('utf-8')` → Convierte bytes de vuelta a string para guardar

**POR QUÉ rounds=12:**
- rounds=10 → 0.1 segundos
- rounds=12 → 0.3 segundos (producción recomendado)
- rounds=15 → 2.5 segundos (muy lento)

### **CREATE: Registrar nuevo usuario**

def register_user(email: str, password: str, nombre: str = None, apellidos: str = None):
    session = SessionLocal()
    try:
        # 1. Verificar que el email no existe
        existing_user = session.query(User).filter(User.email == email).first()
        if existing_user:
            raise ValueError(f"Ya existe un usuario con el email {email}")
        
        # 2. Crear nuevo usuario
        user = User(
            email=email,
            password=User.create_password(password),  # Hash automático
            nombre=nombre,
            apellidos=apellidos,
            is_active=True,
            is_verified=False  # Debe verificar email
        )
        
        # 3. Guardar en base de datos
        session.add(user)
        session.commit()
        session.refresh(user)
        
        print(f"Usuario {user.nombre_completo} registrado exitosamente")
        return user
        
    except IntegrityError as e:
        session.rollback()
        if "email" in str(e.orig):
            raise ValueError("Email ya está en uso")
        else:
            raise ValueError("Error de datos duplicados")
    
    except Exception as e:
        session.rollback()
        raise e
    
    finally:
        session.close()

# USO
new_user = register_user(
    email="juan@email.com",
    password="MiPassword123!",
    nombre="Juan",
    apellidos="Pérez"
)
```

def search_users(query: str, limit: int = 50):
    session = SessionLocal()
    try:
        # Buscar en nombre, apellidos y email
        users = session.query(User)\
            .filter(
                or_(
                    User.nombre.ilike(f"%{query}%"),
                    User.apellidos.ilike(f"%{query}%"),
                    User.email.ilike(f"%{query}%")
                )
            )\
            .filter(User.is_active == True)\
            .order_by(User.nombre, User.apellidos)\
            .limit(limit)\
            .all()
        
        return users
    finally:
        session.close()

def get_inactive_users_since(days: int = 30):
    session = SessionLocal()
    try:
        cutoff_date = datetime.now() - timedelta(days=days)
        
        users = session.query(User)\
            .filter(User.last_login < cutoff_date)\
            .filter(User.is_active == True)\
            .order_by(User.last_login)\
            .all()
        
        return users
    finally:
        session.close()
```

### **UPDATE: Actualizar datos de usuario**
```python
def update_user_profile(user_id: int, **kwargs):
    session = SessionLocal()
    try:
        user = session.query(User).filter(User.id == user_id).first()
        if not user:
            raise ValueError("Usuario no encontrado")
        
        # Campos permitidos para actualizar
        allowed_fields = ['nombre', 'apellidos', 'avatar_url']
        
        for field, value in kwargs.items():
            if field in allowed_fields:
                setattr(user, field, value)
        
        session.commit()
        session.refresh(user)
        
        return user
        
    except Exception as e:
        session.rollback()
        raise e
    finally:
        session.close()

def change_user_password(user_id: int, old_password: str, new_password: str):
    session = SessionLocal()
    try:
        user = session.query(User).filter(User.id == user_id).first()
        if not user:
            raise ValueError("Usuario no encontrado")
        
        # Verificar contraseña actual
        if not user.check_password(old_password):
            raise ValueError("Contraseña actual incorrecta")
        
        # Cambiar contraseña
        user.password = User.create_password(new_password)
        user.reset_login_attempts()  # Resetear intentos por seguridad
        
        session.commit()
        
        print("Contraseña cambiada exitosamente")
        return user
        
    except Exception as e:
        session.rollback()
        raise e
    finally:
        session.close()

def verify_user_email(user_id: int):
    session = SessionLocal()
    try:
        user = session.query(User).filter(User.id == user_id).first()
        if not user:
            raise ValueError("Usuario no encontrado")
        
        user.is_verified = True
        session.commit()
        
        print(f"Email de {user.email} verificado exitosamente")
        return user
        
    except Exception as e:
        session.rollback()
        raise e
    finally:
        session.close()
```

### **DELETE: Desactivar/eliminar usuarios**

def deactivate_user(user_id: int, reason: str = None):
    """Desactiva usuario sin eliminar datos"""
    session = SessionLocal()
    try:
        user = session.query(User).filter(User.id == user_id).first()
        if not user:

def login_user(email: str, password: str):
    """Login con sistema de bloqueo automático"""
    session = SessionLocal()
    try:
        # Intentar autenticar
        user = User.authenticate(session, email, password)
        
        if user:
            return {
                "success": True,
                "user": user.to_dict(),
                "message": "Login exitoso"
            }
        else:
            # Verificar por qué falló
            user = session.query(User)\
                .filter(User.email == email)\
                .filter(User.is_active == True)\
                .first()
            
            if not user:
                return {
                    "success": False,
                    "message": "Email no registrado o usuario inactivo"
                }
            elif user.is_account_locked():
                return {
                    "success": False,
                    "message": f"Cuenta bloqueada por {user.login_attempts} intentos fallidos. Contacta soporte."
                }
            else:
                attempts_left = 5 - user.login_attempts
                return {
                    "success": False,
                    "message": f"Contraseña incorrecta. Te quedan {attempts_left} intentos."
                }
                
    finally:
        session.close()

def get_user_statistics():
    session = SessionLocal()
    try:
        from sqlalchemy import func
        
        stats = session.query(
            func.count(User.id).label('total_users'),
            func.count(User.id).filter(User.is_active == True).label('active_users'),
            func.count(User.id).filter(User.is_verified == True).label('verified_users'),
            func.count(User.id).filter(User.login_attempts >= 5).label('locked_accounts')
        ).first()
        
        return {
            "total_users": stats.total_users,
            "active_users": stats.active_users,
            "verified_users": stats.verified_users,
            "locked_accounts": stats.locked_accounts,
            "verification_rate": (stats.verified_users / stats.total_users * 100) if stats.total_users > 0 else 0
        }
        
    finally:
        session.close()

def get_registration_trends(days: int = 30):
    session = SessionLocal()
    try:
        from sqlalchemy import func
        from datetime import timedelta
        
        cutoff_date = datetime.now() - timedelta(days=days)
        
        trends = session.query(
            func.date(User.created_at).label('date'),
            func.count(User.id).label('registrations')
        )\
        .filter(User.created_at >= cutoff_date)\
        .group_by(func.date(User.created_at))\
        .order_by(func.date(User.created_at))\
        .all()
        
        return [{
            "date": trend.date.isoformat(),
            "registrations": trend.registrations
        } for trend in trends]
        
    finally:
        session.close()
